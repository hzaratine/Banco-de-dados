CREATE TABLE mecanico (
    cpf CHAR(11) PRIMARY KEY,
    nome VARCHAR(100),
    telefone VARCHAR(20),
    data_admissao DATE
);

CREATE TABLE servicos (
    cod_servico INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(100),
    tempo_medio INT,
    descricao VARCHAR(255)
);

CREATE TABLE cliente (
    cpf CHAR(11) PRIMARY KEY,
    nome VARCHAR(100),
    telefone VARCHAR(20),
    email VARCHAR(100)
);

CREATE TABLE veiculo (
    placa VARCHAR(10) PRIMARY KEY,
    cpf_cliente CHAR(11),
    marca VARCHAR(50),
    modelo VARCHAR(50),
    ano INT,
    cor VARCHAR(30),
    FOREIGN KEY (cpf_cliente) REFERENCES cliente(cpf)
);

CREATE TABLE ordem_de_servico (
    idordem INT PRIMARY KEY AUTO_INCREMENT,
    placa VARCHAR(10),
    valor DECIMAL(10,2),
    status VARCHAR(50),
    problemarelatado VARCHAR(255),
    FOREIGN KEY (placa) REFERENCES veiculo(placa)
);

CREATE TABLE servicos_mecanicos (
    cod_mecanico INT PRIMARY KEY AUTO_INCREMENT,
    cpf_mecanico CHAR(11),
    cod_servico INT,
    idordem INT,
    data_inicio DATE,
    data_saida DATE,
    status VARCHAR(50),
    valorunitario DECIMAL(10,2),

    FOREIGN KEY (cpf_mecanico) REFERENCES mecanico(cpf),
    FOREIGN KEY (cod_servico) REFERENCES servicos(cod_servico),
    FOREIGN KEY (idordem) REFERENCES ordem_de_servico(idordem)
);
